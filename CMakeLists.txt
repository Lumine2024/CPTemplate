cmake_minimum_required(VERSION 3.16)
project(CPTemplate CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CP_TEMPLATE_OPTIMIZE_LEVEL "0" CACHE STRING "Optimization level used by CPTemplate tests (0, 1, or 2)")
set_property(CACHE CP_TEMPLATE_OPTIMIZE_LEVEL PROPERTY STRINGS 0 1 2)
if(NOT CP_TEMPLATE_OPTIMIZE_LEVEL MATCHES "^[012]$")
    message(FATAL_ERROR "CP_TEMPLATE_OPTIMIZE_LEVEL must be 0, 1, or 2")
endif()

foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(flag_var "CMAKE_CXX_FLAGS_${config}")
    if(DEFINED ${flag_var})
        if(MSVC)
            string(REGEX REPLACE "(^| )/O[0-2bdisxty]*" " " ${flag_var} "${${flag_var}}")
        else()
            string(REGEX REPLACE "(^| )-O([0-3sg]|fast)" " " ${flag_var} "${${flag_var}}")
        endif()
        string(STRIP "${${flag_var}}" ${flag_var})
    endif()
endforeach()

if(MSVC)
    if(CP_TEMPLATE_OPTIMIZE_LEVEL STREQUAL "0")
        add_compile_options(/Od)
    else()
        add_compile_options(/O${CP_TEMPLATE_OPTIMIZE_LEVEL})
    endif()
else()
    add_compile_options(-O${CP_TEMPLATE_OPTIMIZE_LEVEL})
endif()

include_directories(${CMAKE_SOURCE_DIR}/include)

enable_testing()
add_subdirectory(tests)

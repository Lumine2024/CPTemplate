file(GLOB_RECURSE test_sources CONFIGURE_DEPENDS "*.cpp")

foreach(test_source IN LISTS test_sources)
    get_filename_component(test_name "${test_source}" NAME_WE)
    add_executable("${test_name}" "${test_source}")
    target_include_directories("${test_name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../doctest")
    target_compile_definitions("${test_name}" PRIVATE DOCTEST_CONFIG_IMPLEMENT_WITH_MAIN)
    if(WIN32)
        target_compile_definitions("${test_name}" PRIVATE NOGDI)
    endif()
    add_test(NAME "${test_name}" COMMAND "${test_name}")
endforeach()
